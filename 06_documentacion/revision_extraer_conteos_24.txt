 25:     "jp_sube": [

 28:     "jp_baja": [
 29:         "e2_jp_baja_bus", "e2_jp_baja_priv"

 31:     "tunel_norte": [

 56: def leer_mediciones(archivo):
 57:     datos = {}
 58: 
 59:     for _, elemento in ET.iterparse(
 60:         archivo, events=("end",)
 61:     ):
 62:         if elemento.tag != "interval":
 63:             continue
 64: 
 65:         clave = (
 66:             elemento.get("id"),
 67:             elemento.get("begin"),
 68:             elemento.get("end")
 69:         )
 70: 
 71:         datos[clave] = (
 72:             elemento.get("nVehEntered"),
 73:             elemento.get("meanSpeed"),
 74:             elemento.get("meanOccupancy"),
 75:             elemento.get("maxJamLengthInMeters")
 76:         )
 77: 
 78:         elemento.clear()
 79: 
 80:     return datos

185:                             (bloque, zona, tipo)

187: 
188:                     vistos[zona].update(nuevos)

190:         finally:
